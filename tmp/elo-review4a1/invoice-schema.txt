

class InvoiceAllocation(Input):
    measurement_id: ID
    amount_cents: Money

class InvoiceInput(Input):
    contract_id: ID
    document_type: Literal['nfe','nfse','other']
    number: Short
    series: Annotated[str,StringConstraints(strip_whitespace=True,max_length=30)]=''
    issued_on: date
    amount_cents: Money
    access_key: Annotated[str,StringConstraints(strip_whitespace=True,max_length=100)]|None=None
    notes: Annotated[str,StringConstraints(strip_whitespace=True,max_length=1000)]=''
    allocations: list[InvoiceAllocation]=Field(min_length=1,max_length=200)

    @model_validator(mode='after')
    def document_and_allocations(self):
        self.number=self.number.upper()
        if self.number.isascii() and self.number.isdigit():self.number=str(int(self.number))
        self.series=self.series.upper()
        if self.series.isascii() and self.series.isdigit():self.series=str(int(self.series))
        if self.document_type=='nfe' and not self.series:self.series='0'
        self.access_key=self.access_key.upper() if self.access_key else None
        if self.document_type=='nfe' and self.access_key and not (len(self.access_key)==44 and self.access_key.isascii() and self.access_key.isdigit()):
            raise ValueError('A chave da NF-e deve ter 44 dígitos. A autenticidade fiscal não é validada pelo Elo.')
        if len({a.measurement_id for a in self.allocations})!=len(self.allocations):raise ValueError('Medição repetida.')
        if sum(a.amount_cents for a in self.allocations)!=self.amount_cents:raise ValueError('A soma dos vínculos deve corresponder ao valor total da nota.')
        return self
