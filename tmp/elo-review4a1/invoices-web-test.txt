
test('notas fiscais: consulta, registro e cancelamento explícitos no BFF e no grupo contratual', () => {
  const routes=load('apps/backoffice/src/lib/api-routes.ts');
  for(const [method,url] of [['GET','/contract_invoices'],['POST','/contract_invoices'],['GET','/contract_invoices/measurement-options'],['GET','/contract_invoices/12'],['POST','/contracts/3/invoices/12/cancel']])
    assert.ok(routes.findApiRoute(method,url),method+' '+url);
  for(const [method,url] of [['PUT','/contract_invoices/12'],['DELETE','/contract_invoices/12'],['GET','/contract_invoices/export'],['POST','/contract_invoices/measurement-options']])
    assert.equal(routes.findApiRoute(method,url),undefined);
  assert.equal(routes.upstreamPath('/contract_invoices'),'/contract_invoices/');
  const registry=load('apps/backoffice/src/workspace/registry.ts');
  assert.ok(registry.ROUTINES['ope.notas_fiscais'].hasRecord);
  const groups=load('apps/backoffice/src/workspace/groups.ts');
  assert.equal(groups.groupOf('ope.notas_fiscais','operacao'),'contratos');
});
