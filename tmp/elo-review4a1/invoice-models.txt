

ContractInvoice=register('ContractInvoice','contract_invoices',{
    'contract_id':Column(Integer,nullable=False),'supplier_id':Column(Integer,ForeignKey('suppliers.id'),nullable=False),
    'document_type':Column(String(10),nullable=False),'number':Column(String(80),nullable=False),
    'series':Column(String(30),nullable=False,server_default=text("''")),
    'issued_on':Column(Date,nullable=False),'amount_cents':Column(BigInteger,nullable=False),
    'access_key':Column(String(100)),'notes':Column(String(1000),nullable=False,server_default=text("''")),
    'status':Column(String(20),nullable=False,server_default=text("'registered'")),
    'cancel_reason':Column(String(1000)),'cancelled_by':Column(Integer,ForeignKey('users.id')),
    'cancelled_at':Column(TIMESTAMP(timezone=True)),
},(contract_fk('contract_invoices'),
   CheckConstraint("amount_cents > 0 AND document_type IN ('nfe','nfse','other') AND status IN ('registered','cancelled')",name='ck_contract_invoices_values'),
   CheckConstraint("(status='registered' AND cancel_reason IS NULL AND cancelled_by IS NULL AND cancelled_at IS NULL) OR (status='cancelled' AND length(trim(cancel_reason))>0 AND cancelled_by IS NOT NULL AND cancelled_at IS NOT NULL)",name='ck_contract_invoices_cancel'),
   CheckConstraint("length(trim(number))>0 AND (access_key IS NULL OR length(trim(access_key))>0)",name='ck_contract_invoices_identity'),
   Index('uq_contract_invoice_number','customer_id','supplier_id','document_type','series','number',unique=True,postgresql_where=text("status='registered'")),
   Index('uq_contract_invoice_key','customer_id','access_key',unique=True,postgresql_where=text("status='registered' AND access_key IS NOT NULL"))))

ContractInvoiceLink=register('ContractInvoiceLink','contract_invoice_links',{
    'contract_id':Column(Integer,nullable=False),'invoice_id':Column(Integer,nullable=False),
    'measurement_id':Column(Integer,nullable=False),'amount_cents':Column(BigInteger,nullable=False),
},(contract_fk('contract_invoice_links'),
   ForeignKeyConstraint(['contract_id','invoice_id'],['contract_invoices.contract_id','contract_invoices.id'],name='fk_invoice_link_invoice'),
   ForeignKeyConstraint(['contract_id','measurement_id'],['contract_measurements.contract_id','contract_measurements.id'],name='fk_invoice_link_measurement'),
   UniqueConstraint('invoice_id','measurement_id',name='uq_invoice_link_measurement'),
   CheckConstraint('amount_cents>0',name='ck_invoice_link_amount')))
